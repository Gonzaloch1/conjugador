        /**

         * Muestra un mensaje de error.

         * @param {string} message - El mensaje a mostrar.

        function showError(message) {

            errorText.textContent = message;

            errorMessage.classList.remove('hidden');

            resultsArea.innerHTML = ''; // Limpia resultados previos

        /**

         * Oculta el mensaje de error.

         */

        function hideError() {

            errorMessage.classList.add('hidden');

        }


        /**

         * Valida el verbo ingresado.

         * @param {string} verb - El verbo a validar.

         * @returns {string|null} - La terminación ('ar', 'er', 'ir') o null si es inválido.

         */

        function validateVerb(verb) {

            if (!verb) {

                showError('El campo del verbo no puede estar vacío.');

                return null;

            }

            verb = verb.toLowerCase().trim();

            if (verb.endsWith('ar')) return 'ar';

            if (verb.endsWith('er')) return 'er';

            if (verb.endsWith('ir')) return 'ir';


            showError('El verbo debe terminar en -ar, -er o -ir.');

            return null;

        }


        /**

         * Obtiene la raíz del verbo.

         * @param {string} verb - El verbo en infinitivo.

         * @returns {string} - La raíz del verbo.

         */

        function getStem(verb) {

            return verb.slice(0, -2);

        }


        /**

         * Conjuga un verbo para un tiempo/modo específico.

         * @param {string} stem - La raíz del verbo.

         * @param {string[]} tenseEndings - Las terminaciones para ese tiempo/modo.

         * @returns {string[]} - Un array con las formas conjugadas.

         */

        function conjugate(stem, tenseEndings) {

            // Asegurarse de que tenseEndings es un array

             if (!Array.isArray(tenseEndings)) {

                console.error("Error: tenseEndings no es un array para el stem:", stem);

                return []; // Devuelve array vacío o maneja el error como prefieras

            }

            return tenseEndings.map(ending => stem + ending);

        }


        /**

         * Crea el HTML para una tabla de conjugación.

         * @param {string} title - Título de la tabla.

         * @param {string[]} currentPronouns - Array de pronombres a usar.

         * @param {string[]} conjugatedForms - Array de formas conjugadas.

         * @param {string} description - Descripción del tiempo verbal.

         * @param {string} example - Frase de ejemplo HTML.

         * @returns {string} - El string HTML de la tabla y su descripción.

         */

        function createConjugationTableHTML(title, currentPronouns, conjugatedForms, description, example) {

            let tableRows = '';

            // Asegurarse de que ambos arrays tienen la misma longitud

            const len = Math.min(currentPronouns.length, conjugatedForms.length);

            for (let i = 0; i < len; i++) {

                tableRows += `

                    <tr>

                        <td class="font-medium text-gray-600">${currentPronouns[i]}</td>

                        <td class="text-gray-800">${conjugatedForms[i]}</td>

                    </tr>

                `;

            }

             // Si las longitudes no coinciden, podría ser útil registrar un aviso

            if (currentPronouns.length !== conjugatedForms.length) {

                console.warn(`Discrepancia en longitud para "${title}": ${currentPronouns.length} pronombres vs ${conjugatedForms.length} formas.`);

            }



            return `

                <div class="bg-white p-6 rounded-lg shadow-sm border border-gray-200">

                    <h3 class="text-xl font-semibold text-indigo-700 mb-3">${title}</h3>

                    <p class="text-gray-600 mb-4 text-sm">${description}</p>

                    <p class="text-gray-600 mb-4 text-sm italic">Ej: ${example}</p>

                    <div class="overflow-x-auto">

                        <table class="w-full text-sm conjugation-table">

                            <thead>

                                <tr>

                                    <th>Pronombre</th>

                                    <th>Forma Conjugada</th>

                                </tr>

                            </thead>

                            <tbody>

                                ${tableRows}

                            </tbody>

                        </table>

                    </div>

                </div>

            `;

        }

                    <p class="text-lg text-gray-800 mb-4"><span class="font-medium">Forma:</span> ${verb}</p>


                    <h3 class="text-xl font-semibold text-indigo-700 mb-3">${descGer.title}</h3>

                     <p class="text-gray-600 mb-2 text-sm">${descGer.description}</p>

        // --- Función Principal de Conjugación ---

        function displayConjugations() {

            hideError();

            const verb = verbInput.value.toLowerCase().trim();

            const endingType = validateVerb(verb);


            if (!endingType) {

                // El error ya se mostró en validateVerb

                return;

            }


            const stem = getStem(verb);

            const verbEndings = endings[endingType];

            resultsArea.innerHTML = ''; // Limpia resultados anteriores


            let htmlContent = `<h2 class="text-2xl font-semibold text-gray-700 mb-6 text-center">Conjugación de: <span class="text-indigo-600 font-bold">${verb}</span></h2>`;


            // -- MODO INDICATIVO --

            htmlContent += `<div class="mb-6"><h3 class="text-xl font-bold text-center text-gray-600 mb-4 pb-2 border-b border-gray-300">Modo Indicativo</h3><div class="space-y-6">`;

            const indicativeTenses = [

                'presenteIndicativo', 'imperfectoIndicativo', 'preteritoIndicativo',

                'futuroIndicativo', 'condicionalIndicativo'

            ];

            indicativeTenses.forEach(tense => {

                const conjugated = conjugate(stem, verbEndings[tense]);

                const desc = descriptions[tense];

                // Usar la primera persona (yo) para el ejemplo del indicativo

                const exampleSentence = desc.example(conjugated[0]); // conjugated[0] es la forma 'yo'

                htmlContent += createConjugationTableHTML(desc.title, pronouns.indicative, conjugated, desc.description, exampleSentence);

            });

            htmlContent += `</div></div>`;


            // -- MODO SUBJUNTIVO --

            htmlContent += `<div class="mb-6"><h3 class="text-xl font-bold text-center text-gray-600 mb-4 pb-2 border-b border-gray-300">Modo Subjuntivo</h3><div class="space-y-6">`;

            const subjunctiveTenses = [

                'presenteSubjuntivo', 'imperfectoSubjuntivoRa', 'imperfectoSubjuntivoSe'

            ];

            subjunctiveTenses.forEach(tense => {

                const conjugated = conjugate(stem, verbEndings[tense]);

                 const desc = descriptions[tense];

                 // Usar la primera persona (yo) para el ejemplo del subjuntivo

                const exampleSentence = desc.example(conjugated[0]); // conjugated[0] es la forma 'yo'

                htmlContent += createConjugationTableHTML(desc.title, pronouns.indicative, conjugated, desc.description, exampleSentence);

            });

             htmlContent += `</div></div>`;


            // -- MODO IMPERATIVO --

            htmlContent += `<div class="mb-6"><h3 class="text-xl font-bold text-center text-gray-600 mb-4 pb-2 border-b border-gray-300">Modo Imperativo</h3><div class="space-y-6">`;

            // Afirmativo

            const imperativeAffirmative = conjugate(stem, verbEndings.imperativoAfirmativo);

            const descAff = descriptions.imperativoAfirmativo;

            // Usar la segunda persona (tú) para el ejemplo del imperativo afirmativo

            const exampleAff = descAff.example(imperativeAffirmative[0]); // imperativeAffirmative[0] es la forma 'tú'

            htmlContent += createConjugationTableHTML(descAff.title, pronouns.imperative, imperativeAffirmative, descAff.description, exampleAff);


            // Negativo

            // Conjugar primero las formas base (subjuntivo presente para negativo)

            const imperativeNegativeBase = conjugate(stem, verbEndings.imperativoNegativo);

            // Añadir "no" delante de cada forma para la tabla

            const imperativeNegativeTable = imperativeNegativeBase.map(form => `no ${form}`);

            const descNeg = descriptions.imperativoNegativo;

             // Usar la segunda persona base (tú, sin "no") para el ejemplo del imperativo negativo

            const exampleNeg = descNeg.example(imperativeNegativeBase[0]); // imperativeNegativeBase[0] es la forma 'tú' del subjuntivo presente

            htmlContent += createConjugationTableHTML(descNeg.title, pronouns.imperative, imperativeNegativeTable, descNeg.description, exampleNeg);

            htmlContent += `</div></div>`;



            // -- FORMAS NO PERSONALES --

             htmlContent += `<div class="mb-6"><h3 class="text-xl font-bold text-center text-gray-600 mb-4 pb-2 border-b border-gray-300">Formas no Personales</h3><div class="space-y-6">`;

            const gerund = stem + verbEndings.gerundio;

            const participle = stem + verbEndings.participio;

            htmlContent += createNonPersonalFormsHTML(verb, gerund, participle);

            htmlContent += `</div></div>`;



            resultsArea.innerHTML = htmlContent;

        }

        conjugateButton.addEventListener('click', displayConjugations);
